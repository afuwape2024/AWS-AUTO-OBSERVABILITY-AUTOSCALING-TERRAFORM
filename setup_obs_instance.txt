To set up the auto discovery of instance on Grafana
create IAM role and policy for the prometheus instance
==========================================

sudo nano /etc/prometheus/prometheus.yml
global:
  scrape_interval: 15s

scrape_configs:
  - job_name: "prometheus"
    static_configs:
      - targets: ["localhost:9090"]

  - job_name: "node"

    ec2_sd_configs:
      - region: us-east-2
        port: 9100

    relabel_configs:
      - source_labels: [__meta_ec2_tag_Monitor]
        regex: true
        action: keep

      - source_labels: [__meta_ec2_private_ip]
        target_label: instance


#+++++++++++=====================================
Restart prometheus
sudo systemctl restart prometheus
